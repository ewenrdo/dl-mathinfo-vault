\section{Fonction de répartition}

\subsection{Généralités et premières propriétés}
\definition{
    Soit $\mathbb{P}$ une mesure de probabilité sur $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$. 
    On définit la \textbf{fonction de répartition} $F_\mathbb{P} : \mathbb{R} \to [0,1]$ par : 
    $$F_\mathbb{P}(x) = \mathbb{P}(]-\infty, x])$$
}

\theorem{Théorème}{Caractérisation de l'égalité des mesures}{false}{
    Soient $\mathbb{P}$ et $\mathbb{Q}$ deux mesures de probabilité sur $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$. Alors :
    $$\mathbb{P} = \mathbb{Q} \iff F_\mathbb{P} = F_\mathbb{Q}$$
}
\noindent{\textbf{Preuve :}
    \begin{itemize}
        \item[$\Rightarrow$/] Direct.
        \item[$\Leftarrow$] $\{]-\infty, x], x \in \mathbb{R}\}$ est une classe de parties de $\mathbb{R}$ stable par intersection finies, qui engendre la tribu des boréliens, donc on peut utiliser le lemme d'unicité des mesures.
    \end{itemize}
}


\theorem{Proposition}{Croissance}{false}{
    Soit $\mathbb{P}$ une mesure de probabilité sur $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$. Alors $F_\mathbb{P}$ est croissante.
}
\noindent{\textbf{Preuve :} Si $a < b$, $]-\infty, a] \subset ]-\infty, b]$ et donc
    \[
    F_{\mathbb{P}}(a) = \mathbb{P}(]-\infty, a]) \leq \mathbb{P}(]-\infty, b]) = F_{\mathbb{P}}(b)
    \]

où, pour la dernière égalité on a utilisé la propriété de limite décroissante pour une mesure de probabilités.}

\theorem{Proposition}{Continuité à droite}{false}{
    Soit $\mathbb{P}$ une mesure de probabilité sur $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$. Alors $F_\mathbb{P}$ est continue à droite.
}
\noindent{\textbf{Preuve :} On a $]-\infty, a] = \bigcap_{n \in \mathbb{N}^*} ]-\infty, a + \frac{1}{n}]$, et en utilisant que $F_{\mathbb{P}}$ est croissante pour la première égalité ci-dessous,
    \begin{align*}
    \lim_{x \to a, x > a} F_{\mathbb{P}}(x) &= \lim_{n \to \infty} F_{\mathbb{P}}\left(a + \frac{1}{n}\right) \\
    &= \lim_{n \to \infty} \mathbb{P}\left(]-\infty, a + \frac{1}{n}]\right) = \mathbb{P}(]-\infty, a])
    \end{align*}}


\theorem{Proposition}{Limites aux bornes}{false}{
    Soit $\mathbb{P}$ une mesure de probabilité sur $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$. Alors :
    $$\lim_{x \to -\infty} F_\mathbb{P}(x) = 0$$
    $$\lim_{x \to +\infty} F_\mathbb{P}(x) = 1$$
}

\theorem{Proposition}{Fonction de répartition sur un singleton}{false}{
    Soit $\mathbb{P}$ une mesure de probabilité sur $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$. Il vient 
    $$\mathbb{P}(\{a\}) = F_\mathbb{P}(a) - \lim_{x \to a^-} F_\mathbb{P}(x)$$
}

\remark{En particulier, $F_\mathbb{P}$ est continue en $a \in \mathbb{R}$ si et seulement si $\mathbb{P}(\{a\}) = 0$.}
\remark{$F_\mathbb{P}$ admet un nombre au plus dénombrable de discontinuités.}

\training{Démontrer les deux dernières propositions (limites aux bornes et fonction de répartition sur un singleton).}
\noindent\carreaux{10}

\example{Soit $\mathbb{P}$ telle que \[
F_{\mathbb{P}}(x) = \sum_{k \in \mathbb{N}} \frac{2^k \cdot \exp(-3)}{k!} \indic{\{x \ge k\}} + (1 - \exp(-1)) * (1 - \exp(-x)) \indic{\{x \ge 0\}}
\]

\begin{itemize}
    \item \textbf{Que vaut $\mathbb{P}(\{0\})$ ?} On a :
    $\mathbb{P}(\{0\}) = F_\mathbb{P}(0) - \lim_{x \to 0^-} F_\mathbb{P}(x) = F_\mathbb{P}(0) - 0 = \frac{2^0 \cdot \exp(-3)}{0!} + (1 - \exp(-1)) * (1 - \exp(-0)) = \exp(-3)$
    \item \textbf{Déterminer $\{x \in \mathbb{R} : \mathbb{P}(\{x\}) > 0\}$.} Les seuls points où $\mathbb{P}(\{x\}) > 0$ sont les points $x \in \mathbb{N}$, car pour ces points, $\mathbb{P}(\{x\}) = \frac{2^x \cdot \exp(-3)}{x!}$. Ainsi, $\{x \in \mathbb{R} : \mathbb{P}(\{x\}) > 0\} = \mathbb{N}$.
    \item \textbf{Que vaut $\mathbb{P}([1, 2])$ ?} 

La mesure d'un intervalle fermé $[a, b]$ s'exprime par :
\[
\mathbb{P}([1, 2]) = F_\mathbb{P}(2) - \lim_{x \to 1^-} F_\mathbb{P}(x)
\]

Calculons séparément chaque terme :
\begin{itemize}
    \item Pour $F_\mathbb{P}(2)$, les termes $k \in \{0, 1, 2\}$ contribuent à la somme :
    \[
    F_\mathbb{P}(2) = e^{-3}\left(\frac{2^0}{0!} + \frac{2^1}{1!} + \frac{2^2}{2!}\right) + (1 - e^{-1})(1 - e^{-2}) = 5e^{-3} + (1 - e^{-1})(1 - e^{-2})
    \]
    \item Pour la limite à gauche $\lim_{x \to 1^-} F_\mathbb{P}(x)$, seul le terme $k=0$ intervient (car $x < 1$) :
    \[
    \lim_{x \to 1^-} F_\mathbb{P}(x) = e^{-3}\left(\frac{2^0}{0!}\right) + (1 - e^{-1})(1 - e^{-1}) = e^{-3} + (1 - e^{-1})^2
    \]
\end{itemize}

En faisant la différence, on obtient :
\[
\mathbb{P}([1, 2]) = (5e^{-3} - e^{-3}) + (1 - e^{-1})\Big[(1 - e^{-2}) - (1 - e^{-1})\Big]
\]
\[
\mathbb{P}([1, 2]) = 4e^{-3} + (1 - e^{-1})(e^{-1} - e^{-2}) = 4e^{-3} + e^{-1}(1 - e^{-1})^2
\]
\end{itemize}
}

\remark{Cette fonction de répartition est un mélange d'une loi de Poisson (avec paramètre $\lambda = 3$) et d'une loi exponentielle (avec paramètre $\lambda = 1$).}

\theorem{Théorème}{}{false}{
    Soit $F : \mathbb{R} \to \mathbb{R}$ croissante, continue à droite, telle que $\lim_{x \to -\infty} F(x) = 0$ et $\lim_{x \to +\infty} F(x) = 1$. Alors il existe une unique mesure de probabilité $\mathbb{P}$ sur $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$ telle que $F = F_\mathbb{P}$.
}

\subsection{Densité}

\definition{
    Soit $\mathbb{P}$ une mesure de probabilité sur $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$. Soit $f : \mathbb{R} \to \mathbb{R}_+$ une fonction mesurable.\\
    On dit que $\mathbb{P}$ possède la densité $f$ vis-à-vis de la mesure de Lebesgue si :
    $$\forall A \in \mathcal{B}(\mathbb{R}), \quad \mathbb{P}(A) = \int_A f(x) dx$$
}

\remark{
    Il vient le résultat suivant : $$\int_\mathbb{R} fd\mathbb{P} = \int_\mathbb{R} f \cdot g d\lambda = \int_\mathbb{R} f(x) \cdot g(x) dx$$
    En particulier, si $f\cdot g$ est positive (ou intégrable au sens de Riemann, ie. $\int_\mathbb{R} |f(x)| \cdot g(x) dx < \infty$), $\int_\mathbb{R} f(x)g(x)dx$ coïncide avec l'intégrale de Lebesgue $\int_\mathbb{R} f d\mathbb{P}$.
}

\vocabulary{Par abus de langage, si $X$ est une variable aléatoire de loi $\mathbb{P}_X$, on dit que $X$ possède la densité $f_X$.}

\section{Variables aléatoires}

\definition{
    Soit $(\Omega, \mathcal{F}, \mathbb{P})$ un espace de probabilité. Soit $E$ muni de la tribu $\mathcal{E}$.
    Toute application mesurable $X : \Omega \toMesurable E$ est une variable aléatoire à valeurs dans $E$.\\\\
    La loi de $X$, $\mathbb{P}_X$, est la mesure-image de $\mathbb{P}$ par $X$. On a : $F_X := F_{\mathbb{P}_X}$.
}

\section{Variables aléatoires réelles}
\subsection{Propriétés immédiates}
\remark{Si $E = \mathbb{R}$, on parle de variable aléatoire réelle (v.a.r.). Sa loi $\mathbb{P}_X$ est une mesure de probabilité sur $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$.}

\theorem{Proposition}{Caractérisation par la fonction de répartition}{false}{
    Soit $X$ une variable aléatoire réelle. Alors sa loi $\mathbb{P}_X$ est entièrement caractérisée par sa fonction de répartition $F_X = F_{\mathbb{P}_X}$, définie par :
    $$F_X(a) = \mathbb{P}_X(]-\infty, a]) = \mathbb{P}(X \le a)$$
}

\theorem{Proposition}{Image par une fonction borélienne}{false}{
    Soit $X$ une variable aléatoire réelle et $f : \mathbb{R} \to \mathbb{R}$ une fonction borélienne. Alors $f(X)$ est une variable aléatoire réelle.
}

\theorem{Proposition}{Couple de variables aléatoires}{false}{
    $(X,Y)$ est une variable aléatoire à valeurs dans $\mathbb{R}^2$ si et seulement si $X$ et $Y$ sont des variables aléatoires réelles sur le même espace.\\
    Dans ce cas, $X+Y$ et $X \cdot Y$ sont des variables aléatoires réelles.
}

\theorem{Proposition}{Suite de variables aléatoires}{false}{
    Soit $(X_n)_{n \in \mathbb{N}}$ une suite de variables aléatoires réelles sur un même espace. Alors $\sup_{n \in \mathbb{N}} X_n$, $\inf_{n \in \mathbb{N}} X_n$, $\limsup_{n \to \infty} X_n$ et $\liminf_{n \to \infty} X_n$ sont des variables aléatoires réelles (sous réserve d'existence).
}

\subsection{Espérance, moments et variance}

\reminder{$\mathcal{L}^1(\Omega, \mathcal{F}, \mathbb{P}) = \{ X : \Omega \to \mathbb{R} \text{ v.a.r. mesurable telle que } \int_\Omega |X| \, \mathrm{d}\mathbb{P} < \infty\}$ et $\mathcal{L}^2(\Omega, \mathcal{F}, \mathbb{P}) = \{ X : \Omega \to \mathbb{R} \text{ v.a.r. mesurable telle que } \int_\Omega X^2 \, \mathrm{d}\mathbb{P} < \infty\}$.}
\definition{
    Soit $X$ une v.a.r. positive ou dans $\mathcal{L}^1(\Omega, \mathcal{F}, \mathbb{P})$. On définit l'espérance de $X$ par :
    $$\mathbb{E}[X] = \int_\Omega X d\mathbb{P}$$
}

\definition{
    Soit $X$ dans $\mathcal{L}^2(\Omega, \mathcal{F}, \mathbb{P})$. On définit la variance de $X$ par :
    $$\text{Var}(X) = \mathbb{E}[(X - \mathbb{E}[X])^2] = \begin{cases}
\mathbb{E}[X^2] - (\mathbb{E}[X])^2 & \text{si } \mathbb{E}[|X|] < \infty \\
\infty & \text{sinon}
\end{cases}$$
}

\theorem{Théorème de transfert}{}{false}{
    Soit $\phi : \mathbb{R} \to \mathbb{R}$ une fonction borélienne et $X$ une variable aléatoire réelle. Alors :
    $$\mathbb{E}[\phi(X)] = \int_\Omega \phi(X(\omega)) d\mathbb{P}(\omega) = \int_\mathbb{R} \phi(x) d\mathbb{P}_X(x)$$
    Si $\phi : \mathbb{R} \to \mathbb{R}$ est borélienne, alors $\phi \circ X \in \mathcal{L}^1(\Omega, \mathcal{F}, \mathbb{P})$ si et seulement si $\phi \in \mathcal{L}^1(\mathbb{R}, \mathcal{B}(\mathbb{R}), \mathbb{P}_X)$ et dans ce cas :
    $$\mathbb{E}[\phi(X)] = \int_\Omega \phi(X(\omega)) d\mathbb{P}(\omega) = \int_\mathbb{R} \phi(x) d\mathbb{P}_X(x)$$
}
\ndlr{cf. Laurent pour la preuve (pas encore vue en cours)}

\theorem{Corollaire}{Transfert avec densité}{false}{
    Soit $X$ une variable aléatoire réelle et $f_X$ sa densité vis-à-vis de la mesure de Lebesgue. Alors pour toute fonction borélienne $\phi : \mathbb{R} \to \mathbb{R}$, positive ou telle que $\int_\mathbb{R} |\phi(x)| f_X(x) dx < \infty$, on a :
    $$\mathbb{E}[\phi(X)] = \int_\Omega \phi(X(\omega)) d\mathbb{P}(\omega) = \int_\mathbb{R} \phi(x) f_X(x) dx$$
}

\definition{
    Soit $k \in \mathbb{N}^*$ et $X$ une v.a.r. telle que $X \in \mathcal{L}^k(\Omega, \mathcal{F}, \mathbb{P})$ (c'est-à-dire $\mathbb{E}[|X|^k] < \infty$).
    \begin{itemize}
        \item On appelle \textbf{moment d'ordre $k$} de $X$ le réel :
        $$m_k(X) = \mathbb{E}[X^k]$$
        \item On appelle \textbf{moment factoriel d'ordre $k$} de $X$ le réel $\mathbb{E}\big[X(X-1)\cdots(X-k+1)\big]$.
    \end{itemize}
}

\remark{Sur un espace de probabilité, les espaces $\mathcal{L}^k(\Omega, \mathcal{F}, \mathbb{P})$ sont emboîtés : si $1 \le p \le q$, alors $\mathcal{L}^q(\Omega, \mathcal{F}, \mathbb{P}) \subset \mathcal{L}^p(\Omega, \mathcal{F}, \mathbb{P})$. En particulier, si $X$ admet un moment d'ordre $k$, alors $X$ admet des moments de tout ordre $j \le k$.}

\subsection{Indistinguabilité des variables aléatoires}
\reminder{$\mathcal{L}^1(\Omega, \mathcal{F}, \mathbb{P}) = \{ X : \Omega \to \mathbb{R} \text{ v.a.r. telle que } \mathbb{E}[|X|] < \infty\}$.}

\definition{
    Lorsque $\mathbb{P}(X \neq Y) = \mathbb{P}(\{\omega : X(\omega) \neq Y(\omega)\}) = 0$, 
    on dit que $X$ et $Y$ sont \textbf{indistinguables}. L'indistinguabilité est une relation d'équivalence, notée $\sim$.
}

On travaille à indistinguabilité près quand on se place dans l'espace quotient
\[
\mathbb{L}^1(\Omega, \mathcal{F}, \mathbb{P}) = \mathcal{L}^1(\Omega, \mathcal{F}, \mathbb{P}) / \sim
\]
ou plus généralement pour $p \ge 1$,
\[
\mathbb{L}^p(\Omega, \mathcal{F}, \mathbb{P}) = \{ X \text{ v.a.r. telle que } \mathbb{E}[|X|^p] < \infty \} / \sim.
\]
La norme $\|X\|_p = (\mathbb{E}[|X|^p])^{1/p}$ fait de $\mathbb{L}^p(\Omega, \mathcal{F}, \mathbb{P})$ un espace vectoriel normé complet (espace de Banach).

\theorem{Proposition}{Caractérisation des lois usuelles par leur loi}{false}{
    Soit $X$ une variable aléatoire réelle. Alors :
    \begin{itemize}
        \item $X \sim \mathcal{B}er(p) \iff \mathbb{P}_X = p \delta_1 + (1 - p) \delta_0$ ;
        \item $X \sim \mathcal{B}in(n,p) \iff \mathbb{P}_X = \sum_{k=0}^{n} \binom{n}{k} p^k (1-p)^{n-k} \delta_k$ ;
        \item $X \sim \mathcal{G}eom(p) \iff \mathbb{P}_X = \sum_{k=1}^{\infty} p (1-p)^{k-1} \delta_k$ ;
        \item $X \sim \mathcal{P}oi(\lambda) \iff \mathbb{P}_X = \sum_{k=0}^{\infty} \frac{\lambda^k e^{-\lambda}}{k!} \delta_k$ ;
        \item $X \sim \mathcal{U}(\{1, \ldots, n\}) \iff \mathbb{P}_X = \sum_{k=1}^{n} \frac{1}{n} \delta_k$.
    \end{itemize}
}

\theorem{Proposition}{Transformation affine d'une loi normale}{false}{
    Soient $a \in \mathbb{R}$ et $b \in \mathbb{R}^*$. Si $Z \sim \mathcal{N}(0,1)$, alors la variable aléatoire $X = a + bZ$ suit la loi normale $\mathcal{N}(a, b^2)$.\\
    Réciproquement, si $X \sim \mathcal{N}(\mu, \sigma^2)$ (avec $\sigma > 0$), alors $Z = \dfrac{X - \mu}{\sigma} \sim \mathcal{N}(0,1)$.
}
\noindent{\textbf{Preuve :} Pour toute fonction borélienne positive $\phi : \mathbb{R} \to \mathbb{R}_+$, le théorème de transfert donne :
    $$\mathbb{E}[\phi(X)] = \mathbb{E}[\phi(a + bZ)] = \int_{\mathbb{R}} \phi(a + bz) \frac{1}{\sqrt{2\pi}} e^{-\frac{z^2}{2}} \, dz$$
    En effectuant le changement de variable affine $x = a + bz$ (de sorte que $z = \frac{x-a}{b}$ et $dz = \frac{dx}{|b|}$), on obtient :
    $$\mathbb{E}[\phi(X)] = \int_{\mathbb{R}} \phi(x) \frac{1}{|b|\sqrt{2\pi}} \exp\left(-\frac{(x-a)^2}{2b^2}\right) dx$$
    Par identification de la densité, on conclut que $X \sim \mathcal{N}(a, b^2)$.
}

\subsection{Fonctions génératrices}

\definition{
    Soit $X$ une variable aléatoire à valeurs dans $\mathbb{N}$. On définit la \textbf{fonction génératrice} de $X$, notée $G_X$, sur $[-1, 1]$ par :
    $$G_X(t) = \mathbb{E}[t^X] = \sum_{k=0}^{+\infty} t^k \mathbb{P}(X = k)$$
    La série entière associée a un rayon de convergence $R \ge 1$, ce qui garantit que $G_X$ est de classe $\mathcal{C}^\infty$ sur $]-1, 1[$ et continue sur $[-1, 1]$.
}

\theorem{Proposition}{Caractérisation et lien avec les moments}{false}{
    Soit $X$ une variable aléatoire à valeurs dans $\mathbb{N}$ de fonction génératrice $G_X$.
    \begin{enumerate}
        \item \textbf{Caractérisation de la loi :} $G_X$ caractérise de manière unique la loi de $X$, et pour tout $k \in \mathbb{N}$ :
        $$\mathbb{P}(X = k) = \frac{G_X^{(k)}(0)}{k!}$$
        \item \textbf{Lien avec l'espérance et les moments factoriels :} Pour tout $r \in \mathbb{N}^*$, $X$ admet un moment d'ordre $r$ fini si et seulement si $\lim_{s \to 1^-} G_X^{(r)}(s) < +\infty$, et dans ce cas :
        $$\mathbb{E}\big[X(X-1)\cdots(X-r+1)\big] = \lim_{s \to 1^-} G_X^{(r)}(s) = G_X^{(r)}(1^-)$$
        En particulier, $\mathbb{E}[X] = G_X'(1^-)$ et $\text{Var}(X) = G_X''(1^-) + G_X'(1^-) - \big(G_X'(1^-)\big)^2$.
    \end{enumerate}
}

\subsection{Transformée de Laplace}

\definition{
    Soit $X$ une variable aléatoire réelle. On définit la \textbf{transformée de Laplace} de $X$ (parfois notée $\mathcal{L}_X$ ou $\psi_X$) pour tout $t \in \mathbb{R}$ tel que $\mathbb{E}[e^{-tX}] < \infty$ (ou $\mathbb{E}[e^{tX}] < \infty$ selon la convention) par :
    $$\psi_X(t) = \mathbb{E}[e^{tX}]$$
    Si $X \ge 0$ presque sûrement, $\mathcal{L}_X(t) = \mathbb{E}[e^{-tX}]$ est toujours bien définie et finie sur $\mathbb{R}_+$.
}

\theorem{Proposition}{Propriétés de la transformée de Laplace}{false}{
    \begin{enumerate}
        \item \textbf{Caractérisation de la loi :} Si $\psi_X(t) = \psi_Y(t) < \infty$ pour tout $t \in ]-\varepsilon, \varepsilon[$ ($\varepsilon > 0$), alors $X$ et $Y$ ont même loi ($\mathbb{P}_X = \mathbb{P}_Y$).
        \item \textbf{Transformation affine :} Pour tous $a, b \in \mathbb{R}$, $\psi_{a + bX}(t) = e^{at} \psi_X(bt)$.
        \item \textbf{Lien avec les moments :} Si $\psi_X$ est finie sur un voisinage de $0$, alors $\psi_X$ est $\mathcal{C}^\infty$ au voisinage de $0$, $X$ admet des moments de tout ordre, et :
        $$\forall k \in \mathbb{N}, \quad \mathbb{E}[X^k] = \psi_X^{(k)}(0)$$
    \end{enumerate}
}
